export const formatPrice = (amount: number, currency: string = 'jpy'): string => {
  return new Intl.NumberFormat('ja-JP', {
    style: 'currency',
    currency: currency.toUpperCase(),
    minimumFractionDigits: currency === 'jpy' ? 0 : 2,
  }).format(amount);
};
