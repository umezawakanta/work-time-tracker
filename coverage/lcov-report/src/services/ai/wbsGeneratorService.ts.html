  private calculateCriticalPath(tasks: WBSNode[]): string[] {
    // 簡単なクリティカルパス計算（実際のプロジェクトではより複雑なアルゴリズムが必要）

      .sort((a, b) => (b.estimatedHours || 0) - (a.estimatedHours || 0))

  }
