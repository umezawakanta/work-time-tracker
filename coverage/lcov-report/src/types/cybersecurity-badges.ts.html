  id: string;

  difficulty: 'bronze' | 'silver' | 'gold' | 'platinum' | 'legendary';

  requirements: SecurityRequirement[];
  isUnlocked: boolean;
  progress: number;
  nextMilestone?: string;
  prerequisites?: string[];
  points: number;
  rewards: string[];
  learningPath: LearningModule[];
}

    | 'threat_analysis'
    | 'risk_assessment';
  target: number | string;
  current: number | string;
  description: string;
  progress?: number;
  isCompleted?: boolean;
}
